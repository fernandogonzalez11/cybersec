class stream():
    def __init__(self, seed = os.urandom(8)):
        self.state = hashlib.sha256(str(seed).encode()).digest()[:len(flag)]
    
    def next(self):
        out = self.state[0]
        self.state = self.state[1:] + bytes([(out + 1) % 256])
        return out
    
    def get_bytes(self, num):
        return bytes(self.next()for _ in range(num))
